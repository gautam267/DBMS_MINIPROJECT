/* 1. DEPARTMENT */
CREATE TABLE Department (
    Dept_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Dept_name VARCHAR2(100) NOT NULL,
    HOD_id NUMBER,
    Location VARCHAR2(100)
);

/* 2. FACULTY */
CREATE TABLE Faculty (
    Faculty_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    F_name VARCHAR2(50) NOT NULL,
    L_name VARCHAR2(50),
    Email VARCHAR2(100) UNIQUE,
    Phone_no VARCHAR2(15),
    Designation VARCHAR2(50),
    Dept_id NUMBER,
    CONSTRAINT fk_faculty_dept
        FOREIGN KEY (Dept_id) REFERENCES Department(Dept_id)
);

/* 3. COURSE */
CREATE TABLE Course (
    Course_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Course_name VARCHAR2(100) NOT NULL,
    Duration NUMBER,
    Credits NUMBER,
    Dept_id NUMBER,
    CONSTRAINT fk_course_dept
        FOREIGN KEY (Dept_id) REFERENCES Department(Dept_id)
);

/* 4. SUBJECT */
CREATE TABLE Subject (
    Subject_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Subject_name VARCHAR2(100) NOT NULL,
    Credits NUMBER,
    Type VARCHAR2(30),
    Course_id NUMBER NOT NULL,
    CONSTRAINT fk_subject_course
        FOREIGN KEY (Course_id) REFERENCES Course(Course_id)
);

/* 5. HOSTEL */
CREATE TABLE Hostel (
    Hostel_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Hostel_name VARCHAR2(100),
    Block VARCHAR2(20),
    Room_no VARCHAR2(20),
    Capacity NUMBER
);

/* 6. STUDENT */
CREATE TABLE Student (
    Roll_no NUMBER PRIMARY KEY,
    F_name VARCHAR2(50) NOT NULL,
    L_name VARCHAR2(50),
    DOB DATE,
    Gender VARCHAR2(20),
    Phone_no VARCHAR2(15),
    Email VARCHAR2(100) UNIQUE,
    Address VARCHAR2(200),
    Hostel_id NUMBER,
    CONSTRAINT fk_student_hostel
        FOREIGN KEY (Hostel_id) REFERENCES Hostel(Hostel_id)
);

/* 7. ENROLLMENT */
CREATE TABLE Enrollment (
    Enroll_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Roll_no NUMBER NOT NULL,
    Subject_id NUMBER NOT NULL,
    Semester NUMBER,
    Enroll_date DATE DEFAULT SYSDATE,
    Status VARCHAR2(30),
    CONSTRAINT fk_enroll_student
        FOREIGN KEY (Roll_no) REFERENCES Student(Roll_no),
    CONSTRAINT fk_enroll_subject
        FOREIGN KEY (Subject_id) REFERENCES Subject(Subject_id),
    CONSTRAINT uq_student_subject
        UNIQUE (Roll_no, Subject_id, Semester)
);

/* 8. EXAM */
CREATE TABLE Exam (
    Exam_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Exam_code VARCHAR2(30),
    Exam_name VARCHAR2(100),
    Exam_date DATE,
    Exam_time VARCHAR2(20),
    Max_marks NUMBER,
    Subject_id NUMBER NOT NULL,
    CONSTRAINT fk_exam_subject
        FOREIGN KEY (Subject_id) REFERENCES Subject(Subject_id)
);

/* 9. MARKS */
CREATE TABLE Marks (
    Marks_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Roll_no NUMBER NOT NULL,
    Exam_id NUMBER NOT NULL,
    Internal_marks NUMBER,
    External_marks NUMBER,
    Total_marks NUMBER,
    Grade VARCHAR2(5),
    CONSTRAINT fk_marks_student
        FOREIGN KEY (Roll_no) REFERENCES Student(Roll_no),
    CONSTRAINT fk_marks_exam
        FOREIGN KEY (Exam_id) REFERENCES Exam(Exam_id),
    CONSTRAINT uq_student_exam
        UNIQUE (Roll_no, Exam_id)
);

/* 10. PLACEMENT */
CREATE TABLE Placement (
    Placement_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Roll_no NUMBER NOT NULL,
    Company_name VARCHAR2(100),
    Package NUMBER(10,2),
    Placement_date DATE,
    Grade VARCHAR2(10),
    CONSTRAINT fk_placement_student
        FOREIGN KEY (Roll_no) REFERENCES Student(Roll_no)
);

/* 11. SOFTWARE */
CREATE TABLE Software (
    Software_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Software_name VARCHAR2(100),
    Version VARCHAR2(30),
    License_type VARCHAR2(50)
);

/* 12. FACULTY_SUBJECT: M:N relationship */
CREATE TABLE Faculty_Subject (
    Faculty_id NUMBER,
    Subject_id NUMBER,
    PRIMARY KEY (Faculty_id, Subject_id),
    FOREIGN KEY (Faculty_id) REFERENCES Faculty(Faculty_id),
    FOREIGN KEY (Subject_id) REFERENCES Subject(Subject_id)
);

/* 13. STUDENT_SOFTWARE: M:N relationship */
CREATE TABLE Student_Software (
    Roll_no NUMBER,
    Software_id NUMBER,
    PRIMARY KEY (Roll_no, Software_id),
    FOREIGN KEY (Roll_no) REFERENCES Student(Roll_no),
    FOREIGN KEY (Software_id) REFERENCES Software(Software_id)
);
